-- REVIEWABLE, NOT APPLIED. Additive only. No DROP, no data migration beyond an
-- additive backfill of the binding column from existing approved requests.
--
-- Apply only after Codex review, ideally against a disposable database first:
--   psql "$DISPOSABLE_DB_URL" -f 001_revision_binding_and_email_outbox.sql.txt
--
-- Purpose (review blockers, this pass):
--   1. Bind the ACCEPTED revision request id to the record inside ONE
--      transaction that also enforces eligibility + allowance, so no generation
--      can ever read "the latest approved row by entity".
--   2. Give email delivery real states (claimed / accepted / ambiguous /
--      failed) in a durable outbox, so `preview_sent_at` stops doubling as a
--      claim token and acceptance is never confused with delivery.
-- ============================================================================

-- ---------------------------------------------------------------------------
-- 1. Binding columns (nullable, additive)
-- ---------------------------------------------------------------------------
ALTER TABLE public.leads
  ADD COLUMN IF NOT EXISTS bound_revision_request_id uuid REFERENCES public.revision_requests(id),
  ADD COLUMN IF NOT EXISTS bound_revision_at timestamptz;

ALTER TABLE public.orders
  ADD COLUMN IF NOT EXISTS bound_revision_request_id uuid REFERENCES public.revision_requests(id),
  ADD COLUMN IF NOT EXISTS bound_revision_at timestamptz;

-- Additive backfill: existing in-flight revisions keep working after deploy.
UPDATE public.leads l
   SET bound_revision_request_id = r.id,
       bound_revision_at = r.submitted_at
  FROM (
    SELECT DISTINCT ON (lead_id) id, lead_id, submitted_at
      FROM public.revision_requests
     WHERE lead_id IS NOT NULL AND status = 'approved'
     ORDER BY lead_id, submitted_at DESC
  ) r
 WHERE r.lead_id = l.id
   AND l.bound_revision_request_id IS NULL;

UPDATE public.orders o
   SET bound_revision_request_id = r.id,
       bound_revision_at = r.submitted_at
  FROM (
    SELECT DISTINCT ON (order_id) id, order_id, submitted_at
      FROM public.revision_requests
     WHERE order_id IS NOT NULL AND status = 'approved'
     ORDER BY order_id, submitted_at DESC
  ) r
 WHERE r.order_id = o.id
   AND o.bound_revision_request_id IS NULL;

-- ---------------------------------------------------------------------------
-- 2. Atomic claim + supersede + approve + bind, for BOTH leads and orders.
--    Single statement-level transaction: either the caller owns the revision
--    and the bound request id is durable, or nothing changed at all.
--    Returns: ('claimed' | 'not_eligible' | 'no_allowance' | 'purchased' |
--              'already_bound' | 'missing_request'), plus the bound id.
-- ---------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION public.claim_revision_binding(
  p_entity_type text,
  p_entity_id uuid,
  p_request_id uuid,
  p_expected_revision_count integer,
  p_lead_updates jsonb DEFAULT '{}'::jsonb
)
RETURNS TABLE (result text, bound_request_id uuid, revision_count integer)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path TO 'public'
AS $$
DECLARE
  v_lead public.leads;
  v_order public.orders;
  v_req public.revision_requests;
BEGIN
  -- The request row must already exist and belong to this entity.
  SELECT * INTO v_req FROM public.revision_requests WHERE id = p_request_id FOR UPDATE;
  IF v_req.id IS NULL THEN
    RETURN QUERY SELECT 'missing_request'::text, NULL::uuid, NULL::integer;
    RETURN;
  END IF;

  IF p_entity_type = 'lead' THEN
    IF v_req.lead_id IS DISTINCT FROM p_entity_id THEN
      RETURN QUERY SELECT 'missing_request'::text, NULL::uuid, NULL::integer;
      RETURN;
    END IF;

    -- Row lock serialises concurrent submissions for this lead.
    SELECT * INTO v_lead FROM public.leads WHERE id = p_entity_id FOR UPDATE;
    IF v_lead.id IS NULL THEN
      RETURN QUERY SELECT 'not_eligible'::text, NULL::uuid, NULL::integer;
      RETURN;
    END IF;

    -- Paid / converted in flight: keep the payment, never remake behind it.
    IF v_lead.order_id IS NOT NULL OR v_lead.status = 'converted' THEN
      RETURN QUERY SELECT 'purchased'::text, v_lead.bound_revision_request_id, v_lead.revision_count;
      RETURN;
    END IF;

    -- Another revision already in flight.
    IF lower(coalesce(v_lead.revision_status, '')) IN ('processing', 'pending') THEN
      RETURN QUERY SELECT 'already_bound'::text, v_lead.bound_revision_request_id, v_lead.revision_count;
      RETURN;
    END IF;

    -- Optimistic allowance check against the value the caller read.
    IF coalesce(v_lead.revision_count, 0) IS DISTINCT FROM coalesce(p_expected_revision_count, 0) THEN
      RETURN QUERY SELECT 'not_eligible'::text, v_lead.bound_revision_request_id, v_lead.revision_count;
      RETURN;
    END IF;

    IF coalesce(v_lead.revision_count, 0) >= coalesce(v_lead.max_revisions, 1) THEN
      RETURN QUERY SELECT 'no_allowance'::text, v_lead.bound_revision_request_id, v_lead.revision_count;
      RETURN;
    END IF;

    -- Exactly one approved request per record.
    UPDATE public.revision_requests
       SET status = 'superseded'
     WHERE lead_id = p_entity_id AND status = 'approved' AND id <> p_request_id;

    UPDATE public.revision_requests
       SET status = 'approved', reviewed_at = now(), reviewed_by = 'auto'
     WHERE id = p_request_id;

    UPDATE public.leads
       SET revision_status = 'processing',
           revision_requested_at = now(),
           revision_count = coalesce(revision_count, 0) + 1,
           pending_revision = true,
           bound_revision_request_id = p_request_id,
           bound_revision_at = now()
     WHERE id = p_entity_id
    RETURNING * INTO v_lead;

    RETURN QUERY SELECT 'claimed'::text, p_request_id, v_lead.revision_count;
    RETURN;
  END IF;

  IF p_entity_type = 'order' THEN
    IF v_req.order_id IS DISTINCT FROM p_entity_id THEN
      RETURN QUERY SELECT 'missing_request'::text, NULL::uuid, NULL::integer;
      RETURN;
    END IF;

    SELECT * INTO v_order FROM public.orders WHERE id = p_entity_id FOR UPDATE;
    IF v_order.id IS NULL THEN
      RETURN QUERY SELECT 'not_eligible'::text, NULL::uuid, NULL::integer;
      RETURN;
    END IF;

    IF lower(coalesce(v_order.revision_status, '')) IN ('processing', 'pending') THEN
      RETURN QUERY SELECT 'already_bound'::text, v_order.bound_revision_request_id, v_order.revision_count;
      RETURN;
    END IF;

    IF coalesce(v_order.revision_count, 0) IS DISTINCT FROM coalesce(p_expected_revision_count, 0) THEN
      RETURN QUERY SELECT 'not_eligible'::text, v_order.bound_revision_request_id, v_order.revision_count;
      RETURN;
    END IF;

    IF coalesce(v_order.revision_count, 0) >= coalesce(v_order.max_revisions, 1) THEN
      RETURN QUERY SELECT 'no_allowance'::text, v_order.bound_revision_request_id, v_order.revision_count;
      RETURN;
    END IF;

    UPDATE public.revision_requests
       SET status = 'superseded'
     WHERE order_id = p_entity_id AND status = 'approved' AND id <> p_request_id;

    UPDATE public.revision_requests
       SET status = 'approved', reviewed_at = now(), reviewed_by = 'auto'
     WHERE id = p_request_id;

    UPDATE public.orders
       SET revision_status = 'processing',
           revision_requested_at = now(),
           revision_count = coalesce(revision_count, 0) + 1,
           pending_revision = true,
           bound_revision_request_id = p_request_id,
           bound_revision_at = now()
     WHERE id = p_entity_id
    RETURNING * INTO v_order;

    RETURN QUERY SELECT 'claimed'::text, p_request_id, v_order.revision_count;
    RETURN;
  END IF;

  RETURN QUERY SELECT 'not_eligible'::text, NULL::uuid, NULL::integer;
END;
$$;

REVOKE ALL ON FUNCTION public.claim_revision_binding(text, uuid, uuid, integer, jsonb) FROM PUBLIC;
GRANT EXECUTE ON FUNCTION public.claim_revision_binding(text, uuid, uuid, integer, jsonb) TO service_role;

-- ---------------------------------------------------------------------------
-- 3. Email outbox: real delivery states, service-role only.
--    claimed  -> we own the attempt, nothing sent yet
--    accepted -> the provider accepted the message (delivery attempted once)
--    ambiguous-> request threw/timed out; unknown whether it was sent
--    failed   -> definite provider rejection; safe to retry
-- ---------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS public.email_outbox (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  idempotency_key text NOT NULL,
  entity_type text NOT NULL,
  entity_id uuid NOT NULL,
  purpose text NOT NULL,
  generation_key text,
  state text NOT NULL DEFAULT 'claimed',
  attempt_count integer NOT NULL DEFAULT 1,
  max_attempts integer NOT NULL DEFAULT 3,
  claimed_at timestamptz NOT NULL DEFAULT now(),
  accepted_at timestamptz,
  ambiguous_at timestamptz,
  failed_at timestamptz,
  provider_message_id text,
  last_error text,
  recipients jsonb,
  updated_at timestamptz NOT NULL DEFAULT now()
);

CREATE UNIQUE INDEX IF NOT EXISTS email_outbox_idempotency_key_uidx
  ON public.email_outbox (idempotency_key);
CREATE INDEX IF NOT EXISTS email_outbox_state_idx
  ON public.email_outbox (state, claimed_at);

GRANT ALL ON public.email_outbox TO service_role;
ALTER TABLE public.email_outbox ENABLE ROW LEVEL SECURITY;
-- No anon/authenticated policies on purpose: internal delivery bookkeeping only.

-- Atomic claim: first caller inserts and gets 'claimed'; a caller arriving while
-- another attempt is claimed/accepted gets that state back and MUST NOT send.
-- A previously failed attempt may be retried until max_attempts.
CREATE OR REPLACE FUNCTION public.claim_email_send(
  p_idempotency_key text,
  p_entity_type text,
  p_entity_id uuid,
  p_purpose text,
  p_generation_key text DEFAULT NULL,
  p_recipients jsonb DEFAULT NULL,
  p_max_attempts integer DEFAULT 3
)
RETURNS TABLE (outbox_id uuid, state text, attempt_count integer, claimed boolean)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path TO 'public'
AS $$
DECLARE
  v_row public.email_outbox;
BEGIN
  INSERT INTO public.email_outbox (idempotency_key, entity_type, entity_id, purpose, generation_key, recipients, max_attempts)
  VALUES (p_idempotency_key, p_entity_type, p_entity_id, p_purpose, p_generation_key, p_recipients, coalesce(p_max_attempts, 3))
  ON CONFLICT (idempotency_key) DO NOTHING
  RETURNING * INTO v_row;

  IF v_row.id IS NOT NULL THEN
    RETURN QUERY SELECT v_row.id, v_row.state, v_row.attempt_count, true;
    RETURN;
  END IF;

  SELECT * INTO v_row FROM public.email_outbox WHERE idempotency_key = p_idempotency_key FOR UPDATE;

  IF v_row.state = 'failed' AND v_row.attempt_count < v_row.max_attempts THEN
    UPDATE public.email_outbox
       SET state = 'claimed',
           attempt_count = attempt_count + 1,
           claimed_at = now(),
           failed_at = NULL,
           updated_at = now()
     WHERE id = v_row.id
    RETURNING * INTO v_row;
    RETURN QUERY SELECT v_row.id, v_row.state, v_row.attempt_count, true;
    RETURN;
  END IF;

  RETURN QUERY SELECT v_row.id, v_row.state, v_row.attempt_count, false;
END;
$$;

REVOKE ALL ON FUNCTION public.claim_email_send(text, text, uuid, text, text, jsonb, integer) FROM PUBLIC;
GRANT EXECUTE ON FUNCTION public.claim_email_send(text, text, uuid, text, text, jsonb, integer) TO service_role;

CREATE OR REPLACE FUNCTION public.settle_email_send(
  p_outbox_id uuid,
  p_state text,
  p_provider_message_id text DEFAULT NULL,
  p_error text DEFAULT NULL
)
RETURNS text
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path TO 'public'
AS $$
BEGIN
  IF p_state NOT IN ('accepted', 'ambiguous', 'failed') THEN
    RAISE EXCEPTION 'invalid outbox state %', p_state;
  END IF;

  UPDATE public.email_outbox
     SET state = p_state,
         accepted_at = CASE WHEN p_state = 'accepted' THEN now() ELSE accepted_at END,
         ambiguous_at = CASE WHEN p_state = 'ambiguous' THEN now() ELSE ambiguous_at END,
         failed_at = CASE WHEN p_state = 'failed' THEN now() ELSE failed_at END,
         provider_message_id = coalesce(p_provider_message_id, provider_message_id),
         last_error = p_error,
         updated_at = now()
   WHERE id = p_outbox_id
     AND state = 'claimed';

  RETURN p_state;
END;
$$;

REVOKE ALL ON FUNCTION public.settle_email_send(uuid, text, text, text) FROM PUBLIC;
GRANT EXECUTE ON FUNCTION public.settle_email_send(uuid, text, text, text) TO service_role;

--- state ---------------------------------------------------------------------

  const [payingWithPayPal, setPayingWithPayPal] = useState(false);
  const busy = purchasing || payingWithPayPal;   // mutual lock

--- handler -------------------------------------------------------------------

  const handlePayPalPurchase = async () => {
    if (!token || busy) return;
    setPayingWithPayPal(true);
    try {
      const eligibleSlug = previewData?.targetedPromoEligible
        ? previewData?.targetedPromoSlug ?? null
        : (previewData?.flash20Eligible ? "flash20" : null);

      const response = await fetch(
        `${import.meta.env.VITE_SUPABASE_URL}/functions/v1/create-paypal-lead-order`,
        {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            previewToken: token,
            applyFollowupDiscount: isFollowup,
            applyVday10Discount: isVday10,
            addons: { forever_memory: packageSelected },
            promoSlug: eligibleSlug ?? (promoParam && promoParam !== "flash20" ? promoParam : undefined),
          }),
        }
      );
      const data = await response.json();

      if (!response.ok) {
        if (data.error === "paypal_unavailable") {
          toast({
            title: "PayPal isn't available right now",
            description: "You can complete your purchase with a card instead.",
          });
          setPayingWithPayPal(false);
          return;
        }
        if (data.error === "package_not_ready") {
          setPackageSelected(false);
          toast({
            title: "The package is still getting ready",
            description: "Your full song is ready now. The extra version appears on your song page once it finishes.",
          });
          setPayingWithPayPal(false);
          return;
        }
        throw new Error(data.error || "Could not start PayPal");
      }

      window.location.href = `https://www.paypal.com/checkoutnow?token=${data.orderID}`;
    } catch (e) {
      console.error("PayPal checkout error:", e);
      toast({
        title: "Couldn't reach PayPal",
        description: "Please try again, or pay with a card.",
        variant: "destructive",
      });
      setPayingWithPayPal(false);
    }
  };

--- markup (directly below the existing purchase button, same card) ------------

  {previewData?.paypalEnabled && (
